\section*{Problem 1. Outlining Proofs}

\textbf{(a),(b)}State Machine:
The set of state is $Q = \{ \text{All possible permutations of } \{0,1,2,...,{n-1}\} \}$

\textbf{Start state:} Any one of the permutations of $\{0,1,2,...,{n-1}\}$.

\textbf{Transition:}
\[ (a_0,\dots ,a_i,a_{i+1} \dots a_{j-1},a_j,a_{j+1},\dots,a_{n-1}) \rightarrow (a_0,\dots ,a_j,a_{j-1} \dots a_{i+1},a_i,a_{j+1},\dots,a_{n-1}) \]
IFF $(i, j)$ is an inverted pair (i.e., $i < j$ but $a_i > a_j$).

\textbf{Fianl state:}
$q_f = (a_0,a_1, \dots ,a_{n-1})\text{ } |\text{ }(a_0 < a_1 < \dots < a_{n-1})$ which also implies that the list is sorted.


\noindent\textbf{(c)} Derived variable: For any state
$q = (a_0, a_1, \ldots, a_{n-1})$, define
\[
X(q) := a_0 \Vert a_1 \Vert \cdots \Vert a_{n-2} \Vert a_{n-1}.
\]
where $\Vert$ is the concatenation operator.

Also, $X(q)$ has only a nonnegative integer value because each term of $X(q)$ is part of state q, By definition
each member of every valid state is defined over natural numbers, so the number obtained from concatenating natural numbers must also be natural.

We know that in any number, digits on the left are more significant than digits to the right of them, and during transition we swap terms $a_i$ and $a_j$
where $i < j$ and $a_i > a_j$, so after the swap, the value of the leftmost term taking part in the swap, which is at position $i$, will decrease, resulting in the overall value
of the final concatenated value decreasing. This means $X(q)$ strictly decreases with each transition.

\vspace{\baselineskip}
\noindent\textbf{(d)}:

For the derived variable $X(q)$ , from part (c), we know that its value is defined over natural numbers and $X(q)$ is strictly decreasing.

Then, by the  termination theorem, we can conclude that the  state machine will terminate, and
from part(b) we know that the  final state will be a fully sorted list.

\section*{Problem 4. Some Sums}

\textbf{(a)}
\[
\begin{aligned} 
\sum_{i = 0}^{n}\frac{5^i - 7^i}{13^i} 
&= \sum_{i = 0}^{n} \left(\frac{5}{13}\right)^i - \sum_{i = 0}^{n} \left(\frac{7}{13}\right)^i \\ 
&= \frac{1 - \left(\frac{5}{13}\right)^{n+1}}{1 - \frac{5}{13}} - \frac{1 - \left(\frac{7}{13}\right)^{n+1}}{1 - \frac{7}{13}} \qquad \left( \text{using } \sum_{i=0}^{n} r^i = \frac{1-r^{n+1}}{1-r} \right) \\
&= \frac{13}{8} \left[ 1 - \left(\frac{5}{13}\right)^{n+1} \right] - \frac{13}{6} \left[ 1 - \left(\frac{7}{13}\right)^{n+1} \right] \\
&= -\frac{13}{24} - \frac{13}{8}\left(\frac{5}{13}\right)^{n+1} + \frac{13}{6}\left(\frac{7}{13}\right)^{n+1} 
\end{aligned}
\]

\vspace{\baselineskip}
\textbf{(b)}
\[
\begin{aligned}
\sum_{i=1}^{n} \sum_{j=1}^{k} (3^j - i) 
&= \sum_{i=1}^{n} \sum_{j=1}^{k} 3^j - \sum_{i=1}^{n} \sum_{j=1}^{k} i \\
&= n \sum_{j=1}^{k} 3^j - k \sum_{i=1}^{n} i \qquad \text{(evaluating the outer and inner constant sums)} \\
&= n \left( \frac{3(3^k - 1)}{3 - 1} \right) - k \left( \frac{n(n+1)}{2} \right) \qquad \text{(applying GP and AP series formulas)} \\
&= \frac{3n(3^k - 1)}{2} - \frac{kn(n+1)}{2}
\end{aligned}
\]
